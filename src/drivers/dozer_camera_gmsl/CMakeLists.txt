cmake_minimum_required(VERSION 3.8)
project(dozer_camera_gmsl)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
  add_compile_options(-Wall -Wextra -Wpedantic)
endif()

find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(dozer_media_vendor REQUIRED)

add_executable(camera_gmsl_node src/camera_node.cpp)
target_include_directories(camera_gmsl_node PRIVATE include)
target_link_libraries(camera_gmsl_node dozer_media_vendor::yuv)
ament_target_dependencies(camera_gmsl_node rclcpp sensor_msgs)
install(TARGETS camera_gmsl_node DESTINATION lib/${PROJECT_NAME})

if(BUILD_TESTING)
  enable_testing()
  add_executable(frame_rate_limiter_tests test/frame_rate_limiter_tests.cpp)
  target_include_directories(frame_rate_limiter_tests PRIVATE include)
  add_test(NAME frame_rate_limiter_tests COMMAND frame_rate_limiter_tests)
  add_executable(camera_conversion_tests test/camera_conversion_tests.cpp)
  target_include_directories(camera_conversion_tests PRIVATE include)
  target_link_libraries(camera_conversion_tests dozer_media_vendor::yuv)
  add_test(NAME camera_conversion_tests COMMAND camera_conversion_tests)
endif()

ament_package()
